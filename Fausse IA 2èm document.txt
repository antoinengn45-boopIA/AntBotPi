/* Intègre le core mock (coller la version JS fournie plus haut) */
const JSON4517 = (() => {
  const id = (p='it') => `${p}_${Date.now()}_${Math.floor(Math.random()*1000)}`;
  const makeMeta = (k,l,t) => ({ id: id(k), kind:k, length:l, tone:t, fictional:true, generatedAt:new Date().toISOString() });

  function generateStory({ prompt='', length=100, tone='neutre' }={}) {
    const meta = makeMeta('story', length, tone);
    const text = `[[Récit fictif — ${meta.id}]] ${prompt || 'Une scène vide...'} — (extrait narratif, ton: ${tone}).`;
    return { meta, text };
  }

  function createCharacter({ archetype='détective', style='cyberpunk' }={}) {
    const name = `${archetype}-${Math.floor(Math.random()*900)}`;
    const profile = { id: id('char'), name, archetype, style, traits: ['intriguant','taciturne'], backstory: `Origine: ${style}.`};
    return { meta: makeMeta('character', null, style), profile };
  }

  function fakeDiagnose({ symptoms='étrange', tone='playful' }={}) {
    const diagnosis = `Syndrome de ${symptoms.split(' ')[0] || 'l\'abstrait'}`;
    return { meta: makeMeta('diagnosis', null, tone), diagnosis, advice: 'Ceci est fictif. Ne pas suivre.', fictional: true };
  }

  async function mockFetch(endpoint, body) {
    await new Promise(r=>setTimeout(r,200+Math.random()*400));
    if (endpoint.endsWith('/generate/story')) return generateStory(body);
    if (endpoint.endsWith('/personality/create')) return createCharacter(body);
    if (endpoint.endsWith('/diagnose/fake')) return fakeDiagnose(body);
    return { error: 'endpoint inconnu', endpoint };
  }

  return { generateStory, createCharacter, fakeDiagnose, mockFetch };
})();

/* UI wiring */
const out = document.getElementById('output');
const promptEl = document.getElementById('prompt');

document.getElementById('genStory').onclick = async () => {
  const prompt = promptEl.value;
  const resp = await JSON4517.mockFetch('https://api.j-son4517.mock/v1/generate/story', { prompt, length:120, tone:'lyrique' });
  out.textContent = JSON.stringify(resp, null, 2);
};

document.getElementById('genChar').onclick = async () => {
  const resp = await JSON4517.mockFetch('https://api.j-son4517.mock/v1/personality/create', { archetype:'détective', style:'noir-tech' });
  out.textContent = JSON.stringify(resp, null, 2);
};

document.getElementById('diag').onclick = async () => {
  const resp = await JSON4517.mockFetch('https://api.j-son4517.mock/v1/diagnose/fake', { symptoms:'tousser devant des guirlandes', tone:'playful' });
  out.textContent = JSON.stringify(resp, null, 2);
};
